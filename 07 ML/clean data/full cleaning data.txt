import pandas as pd

# CSV ফাইল লোড
df = pd.read_csv('titanic.csv')

# ----------------------------
# 1. Handle Missing Values
# ----------------------------
# Age এর missing value median দিয়ে fill করা
df['Age'] = df['Age'].fillna(df['Age'].median())

# Embarked এর missing value mode দিয়ে fill করা
df['Embarked'] = df['Embarked'].fillna(df['Embarked'].mode()[0])

# Cabin অনেক missing → drop করা (optional)
df = df.drop(columns=['Cabin'])

# ----------------------------
# 2. Standardize Sex column
# ----------------------------
df['Sex'] = df['Sex'].str.lower().map({'male':0, 'female':1})

# ----------------------------
# 3. One-Hot Encode Embarked
# ----------------------------
df = pd.get_dummies(df, columns=['Embarked'], drop_first=True)  # drop_first=True avoids multicollinearity

# ----------------------------
# 4. Extract Title from Name
# ----------------------------
df['Title'] = df['Name'].str.extract(' ([A-Za-z]+)\.', expand=False)

# Standardize titles
df['Title'] = df['Title'].replace(['Mlle','Ms'],'Miss')
df['Title'] = df['Title'].replace('Mme','Mrs')
df['Title'] = df['Title'].replace(['Lady','Countess','Capt','Col','Don','Dr','Major',
                                   'Rev','Sir','Jonkheer','Dona'],'Rare')

# One-Hot Encode Title
df = pd.get_dummies(df, columns=['Title'], drop_first=True)

# ----------------------------
# 5. Drop unnecessary columns
# ----------------------------
df = df.drop(columns=['Name','Ticket','PassengerId'])

# ----------------------------
# 6. Final check
# ----------------------------
print(df.info())
print(df.head())
